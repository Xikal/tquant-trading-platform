```python
        quote_payloads = await asyncio.gather(
            *(loop.run_in_executor(pool, fetch_one, row) for row in rows)
```

```python
def compute_cvar(returns: list[float], confidence: float = 0.95) -> float:
    """
    CVaR（条件在险价值）：最差 (1-confidence)% 交易的均值亏损。
    
    confidence=0.95 时，取最差的 5% 交易的平均收益率。
    比 avg_loss 更能反映极端情况下的真实风险。
    """
    if not returns:
        return 0.0
    sorted_returns = sorted(returns)
    tail_count = max(1, int(len(sorted_returns) * (1 - confidence)))
    tail = sorted_returns[:tail_count]
    return round(sum(tail) / len(tail), 4)
```

```python
import random

def compute_pbo(
    real_sharpe: float,
    signal_series: list[bool],
    returns_series: list[float],
    n_permutations: int = 1000,
) -> dict:
    """
    PBO（Probability of Better Outcome）：策略是否真的优于随机。
    
    将信号序列随机打乱 n 次，每次重新计算夏普比率。
    统计伪策略中夏普 >= 真实夏普的比例。
    PBO = 1 - 该比例。PBO > 0.9 可认为策略有真实 alpha。
    PBO < 0.5 → 策略大概率是过拟合的噪声。
    """
    random.seed(42)
    pseudo_sharpes = []
    n = len(signal_series)
    
    for _ in range(n_permutations):
        shuffled_idx = list(range(n))
        random.shuffle(shuffled_idx)
        pseudo_signals = [signal_series[i] for i in shuffled_idx]
        pseudo_returns = [
            returns_series[i] if pseudo_signals[i] else 0.0
            for i in range(n)
        ]
        pseudo_sharpe = _compute_sharpe_from_returns(pseudo_returns)
        pseudo_sharpes.append(pseudo_sharpe)
    
    better_count = sum(1 for s in pseudo_sharpes if s >= real_sharpe)
    pbo = 1.0 - better_count / n_permutations
    
    return {
        "pbo": round(pbo, 4),
        "real_sharpe": round(real_sharpe, 4),
        "pseudo_sharpe_mean": round(sum(pseudo_sharpes) / len(pseudo_sharpes), 4),
        "pseudo_sharpe_95pct": round(sorted(pseudo_sharpes)[int(n_permutations * 0.95)], 4),
        "n_permutations": n_permutations,
        "verdict": (
            "likely_overfit" if pbo < 0.5
            else "potentially_valid" if pbo < 0.9
            else "strong_evidence"
        ),
    }


def _compute_sharpe_from_returns(returns: list[float], risk_free: float = 0.02) -> float:
    if not returns:
        return 0.0
    mean = sum(returns) / len(returns)
    variance = sum((r - mean) ** 2 for r in returns) / max(len(returns) - 1, 1)
    std = variance ** 0.5
    if std == 0:
        return 0.0
    return (mean - risk_free / 252) / std
```

### 新增文件

```python
from dataclasses import dataclass

@dataclass
class KellyPosition:
    full_kelly: float
    half_kelly: float
    quarter_kelly: float
    win_rate: float
    avg_win_pct: float
    avg_loss_pct: float
    expected_value: float


def compute_kelly_position(
    win_rate: float,
    avg_win_pct: float,
    avg_loss_pct: float,
    max_position_pct: float = 0.25,
    min_position_pct: float = 0.05,
) -> KellyPosition:
    """
    凯利公式：f = (b*p - q) / b
    
    b = 盈亏比 = avg_win / abs(avg_loss)
    p = 胜率, q = 1-p
    
    半凯利 = full_kelly / 2（推荐实盘使用）
    """
    if avg_loss_pct >= 0 or avg_win_pct <= 0:
        return KellyPosition(0, 0, 0, win_rate, avg_win_pct, avg_loss_pct, 0)
    
    b = avg_win_pct / abs(avg_loss_pct)
    q = 1.0 - win_rate
    
    full_kelly = (b * win_rate - q) / b
    full_kelly = max(0.0, min(full_kelly, max_position_pct))
    
    half_kelly = full_kelly / 2.0
    quarter_kelly = full_kelly / 4.0
    
    if half_kelly < min_position_pct and full_kelly > 0:
        half_kelly = min_position_pct
        quarter_kelly = min_position_pct / 2.0
    
    ev = win_rate * avg_win_pct - q * abs(avg_loss_pct)
    
    return KellyPosition(
        full_kelly=round(full_kelly, 4),
        half_kelly=round(half_kelly, 4),
        quarter_kelly=round(quarter_kelly, 4),
        win_rate=win_rate,
        avg_win_pct=avg_win_pct,
        avg_loss_pct=avg_loss_pct,
        expected_value=round(ev, 4),
    )
```

```python
import random

def simulate_liquidity_crisis(
    filled_signals: list[dict],
    crisis_ratio: float = 0.05,
    crisis_extra_loss_pct: float = -25.0,
) -> dict:
    """
    模拟极端流动性枯竭场景：
    随机选取 crisis_ratio 比例的持仓，假设遇到连续跌停无法卖出，
    每笔额外亏损 crisis_extra_loss_pct。
    """
    if not filled_signals:
        return {
            "scenario": "liquidity_crisis",
            "crisis_count": 0,
            "original_avg_return": 0.0,
            "crisis_avg_return": 0.0,
            "impact_pct": 0.0,
        }
    
    n_crisis = max(1, int(len(filled_signals) * crisis_ratio))
    crisis_indices = random.sample(range(len(filled_signals)), min(n_crisis, len(filled_signals)))
    
    original_returns = [s.get("net_return_pct", 0) for s in filled_signals]
    original_avg = sum(original_returns) / len(original_returns)
    
    adjusted_returns = list(original_returns)
    for i in crisis_indices:
        adjusted_returns[i] += crisis_extra_loss_pct
    
    crisis_avg = sum(adjusted_returns) / len(adjusted_returns)
    
    return {
        "scenario": "liquidity_crisis",
        "crisis_count": n_crisis,
        "crisis_ratio_pct": round(crisis_ratio * 100, 1),
        "crisis_extra_loss_pct": crisis_extra_loss_pct,
        "original_avg_return": round(original_avg, 3),
        "crisis_avg_return": round(crisis_avg, 3),
        "impact_pct": round(crisis_avg - original_avg, 3),
    }
```
